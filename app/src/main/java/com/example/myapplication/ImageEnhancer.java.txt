package com.example.myapplication;

import android.graphics.Bitmap;
import android.util.Log;

import org.opencv.android.Utils;
import org.opencv.core.Core;
import org.opencv.core.CvType;
import org.opencv.core.Mat;
import org.opencv.core.Size;
import org.opencv.imgproc.Imgproc;

import java.io.File;
import java.io.FileOutputStream;
import java.text.SimpleDateFormat;
import java.util.ArrayList;
import java.util.Date;
import java.util.List;
import java.util.Locale;

/**
 * ImageEnhancer - Advanced image processing filters using OpenCV
 * Provides various enhancement filters for document images
 */
public class ImageEnhancer {

    private static final String TAG = "ImageEnhancer";
    private static final String FILENAME_FORMAT = "yyyy-MM-dd-HH-mm-ss-SSS";

    /**
     * Enhancement filter types
     */
    public enum FilterType {
        AUTO_ENHANCE,
        BLACK_AND_WHITE,
        BRIGHTNESS_CONTRAST,
        NOISE_REDUCTION,
        SHARPEN,
        GRAYSCALE,
        ORIGINAL
    }

    /**
     * Apply brightness and contrast adjustment
     *
     * @param inputMat Input image
     * @param brightness Brightness value (-100 to 100, 0 = no change)
     * @param contrast Contrast value (-100 to 100, 0 = no change)
     * @return Enhanced image
     */
    public static Mat adjustBrightnessContrast(Mat inputMat, int brightness, int contrast) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            Mat output = new Mat();

            // Convert brightness and contrast to scale factors
            double alpha = (contrast + 100.0) / 100.0; // Contrast scale (0.0 to 2.0)
            double beta = brightness; // Brightness offset

            // Apply: output = alpha * input + beta
            inputMat.convertTo(output, -1, alpha, beta);

            Log.d(TAG, "Brightness/Contrast adjusted: brightness=" + brightness + ", contrast=" + contrast);
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error adjusting brightness/contrast", e);
            return inputMat.clone();
        }
    }

    /**
     * Auto-enhance image using histogram equalization
     * Improves contrast and visibility automatically
     *
     * @param inputMat Input image
     * @return Auto-enhanced image
     */
    public static Mat autoEnhance(Mat inputMat) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            Mat output = new Mat();

            // Check if image is grayscale or color
            if (inputMat.channels() == 1) {
                // Grayscale image - apply histogram equalization directly
                Imgproc.equalizeHist(inputMat, output);
            } else {
                // Color image - convert to LAB, equalize L channel, convert back
                Mat lab = new Mat();
                Imgproc.cvtColor(inputMat, lab, Imgproc.COLOR_BGR2Lab);

                // Split channels
                List<Mat> labChannels = new ArrayList<>();
                Core.split(lab, labChannels);

                // Equalize L channel
                Imgproc.equalizeHist(labChannels.get(0), labChannels.get(0));

                // Merge channels
                Core.merge(labChannels, lab);

                // Convert back to BGR
                Imgproc.cvtColor(lab, output, Imgproc.COLOR_Lab2BGR);

                // Release intermediate mats
                lab.release();
                for (Mat channel : labChannels) {
                    channel.release();
                }
            }

            Log.d(TAG, "Auto-enhancement applied using histogram equalization");
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error auto-enhancing image", e);
            return inputMat.clone();
        }
    }

    /**
     * Convert image to black and white with adaptive threshold
     *
     * @param inputMat Input image
     * @param threshold Threshold value (0-255), use -1 for adaptive
     * @return Black and white image
     */
    public static Mat convertToBlackAndWhite(Mat inputMat, int threshold) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            Mat gray = new Mat();
            Mat output = new Mat();

            // Convert to grayscale if needed
            if (inputMat.channels() > 1) {
                Imgproc.cvtColor(inputMat, gray, Imgproc.COLOR_BGR2GRAY);
            } else {
                gray = inputMat.clone();
            }

            if (threshold == -1) {
                // Adaptive threshold (best for documents with varying lighting)
                Imgproc.adaptiveThreshold(
                    gray,
                    output,
                    255,
                    Imgproc.ADAPTIVE_THRESH_GAUSSIAN_C,
                    Imgproc.THRESH_BINARY,
                    11,  // Block size
                    2    // Constant subtracted from mean
                );
                Log.d(TAG, "Applied adaptive threshold for B&W conversion");
            } else {
                // Fixed threshold
                Imgproc.threshold(gray, output, threshold, 255, Imgproc.THRESH_BINARY);
                Log.d(TAG, "Applied fixed threshold: " + threshold);
            }

            gray.release();
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error converting to black and white", e);
            return inputMat.clone();
        }
    }

    /**
     * Apply noise reduction filter
     * Uses bilateral filter to reduce noise while preserving edges
     *
     * @param inputMat Input image
     * @param strength Filter strength (1-10, higher = more noise reduction)
     * @return Noise-reduced image
     */
    public static Mat reduceNoise(Mat inputMat, int strength) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            Mat output = new Mat();

            // Clamp strength to valid range
            strength = Math.max(1, Math.min(10, strength));

            // Calculate filter parameters based on strength
            int d = 5 + (strength * 2);  // Diameter of pixel neighborhood (5-25)
            double sigmaColor = 50.0 + (strength * 10.0);  // Filter sigma in color space
            double sigmaSpace = 50.0 + (strength * 10.0);  // Filter sigma in coordinate space

            // Apply bilateral filter
            Imgproc.bilateralFilter(inputMat, output, d, sigmaColor, sigmaSpace);

            Log.d(TAG, "Noise reduction applied: strength=" + strength +
                       ", d=" + d + ", sigmaColor=" + sigmaColor);
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error reducing noise", e);
            return inputMat.clone();
        }
    }

    /**
     * Apply sharpening filter
     * Enhances edges and details in the image
     *
     * @param inputMat Input image
     * @param strength Sharpening strength (1-10, higher = more sharpening)
     * @return Sharpened image
     */
    public static Mat sharpenImage(Mat inputMat, int strength) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            // Clamp strength to valid range
            strength = Math.max(1, Math.min(10, strength));

            // Create blurred version
            Mat blurred = new Mat();
            Size kernelSize = new Size(5, 5);
            Imgproc.GaussianBlur(inputMat, blurred, kernelSize, 1.0);

            // Calculate sharpening amount based on strength
            double alpha = 1.0 + (strength * 0.3);  // Original image weight (1.0 to 4.0)
            double beta = -(strength * 0.3);        // Blurred image weight (0 to -3.0)

            // Apply unsharp mask: sharpened = original * alpha + blurred * beta
            Mat output = new Mat();
            Core.addWeighted(inputMat, alpha, blurred, beta, 0, output);

            blurred.release();

            Log.d(TAG, "Sharpening applied: strength=" + strength +
                       ", alpha=" + alpha + ", beta=" + beta);
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error sharpening image", e);
            return inputMat.clone();
        }
    }

    /**
     * Convert image to grayscale
     *
     * @param inputMat Input image
     * @return Grayscale image
     */
    public static Mat convertToGrayscale(Mat inputMat) {
        if (inputMat == null || inputMat.empty()) {
            Log.e(TAG, "Input image is null or empty");
            return inputMat;
        }

        try {
            if (inputMat.channels() == 1) {
                // Already grayscale
                return inputMat.clone();
            }

            Mat output = new Mat();
            Imgproc.cvtColor(inputMat, output, Imgproc.COLOR_BGR2GRAY);

            Log.d(TAG, "Converted to grayscale");
            return output;

        } catch (Exception e) {
            Log.e(TAG, "Error converting to grayscale", e);
            return inputMat.clone();
        }
    }

    /**
     * Apply enhancement filter by type
     *
     * @param inputBitmap Input bitmap
     * @param filterType Type of filter to apply
     * @return Enhanced bitmap
     */
    public static Bitmap applyFilter(Bitmap inputBitmap, FilterType filterType) {
        if (inputBitmap == null) {
            Log.e(TAG, "Input bitmap is null");
            return null;
        }

        if (!OpenCVHelper.isInitialized()) {
            Log.e(TAG, "OpenCV not initialized");
            return inputBitmap;
        }

        try {
            // Convert bitmap to Mat
            Mat inputMat = OpenCVHelper.bitmapToMat(inputBitmap);
            Mat outputMat;

            // Apply selected filter
            switch (filterType) {
                case AUTO_ENHANCE:
                    outputMat = autoEnhance(inputMat);
                    break;

                case BLACK_AND_WHITE:
                    outputMat = convertToBlackAndWhite(inputMat, -1); // Adaptive threshold
                    break;

                case BRIGHTNESS_CONTRAST:
                    outputMat = adjustBrightnessContrast(inputMat, 10, 20); // Default values
                    break;

                case NOISE_REDUCTION:
                    outputMat = reduceNoise(inputMat, 5); // Medium strength
                    break;

                case SHARPEN:
                    outputMat = sharpenImage(inputMat, 5); // Medium strength
                    break;

                case GRAYSCALE:
                    outputMat = convertToGrayscale(inputMat);
                    break;

                case ORIGINAL:
                default:
                    outputMat = inputMat.clone();
                    break;
            }

            // Convert back to bitmap
            Bitmap outputBitmap = OpenCVHelper.matToBitmap(outputMat);

            // Clean up
            inputMat.release();
            outputMat.release();

            Log.d(TAG, "Filter applied successfully: " + filterType);
            return outputBitmap;

        } catch (Exception e) {
            Log.e(TAG, "Error applying filter: " + filterType, e);
            return inputBitmap;
        }
    }

    /**
     * Apply custom enhancement with all parameters
     *
     * @param inputBitmap Input bitmap
     * @param brightness Brightness adjustment (-100 to 100)
     * @param contrast Contrast adjustment (-100 to 100)
     * @param sharpness Sharpness level (0 to 10)
     * @param noiseReduction Noise reduction level (0 to 10)
     * @return Enhanced bitmap
     */
    public static Bitmap applyCustomEnhancement(Bitmap inputBitmap,
                                                int brightness,
                                                int contrast,
                                                int sharpness,
                                                int noiseReduction) {
        if (inputBitmap == null) {
            Log.e(TAG, "Input bitmap is null");
            return null;
        }

        if (!OpenCVHelper.isInitialized()) {
            Log.e(TAG, "OpenCV not initialized");
            return inputBitmap;
        }

        try {
            Mat mat = OpenCVHelper.bitmapToMat(inputBitmap);

            // Apply filters in sequence
            if (noiseReduction > 0) {
                Mat denoised = reduceNoise(mat, noiseReduction);
                mat.release();
                mat = denoised;
            }

            if (brightness != 0 || contrast != 0) {
                Mat adjusted = adjustBrightnessContrast(mat, brightness, contrast);
                mat.release();
                mat = adjusted;
            }

            if (sharpness > 0) {
                Mat sharpened = sharpenImage(mat, sharpness);
                mat.release();
                mat = sharpened;
            }

            Bitmap result = OpenCVHelper.matToBitmap(mat);
            mat.release();

            Log.d(TAG, "Custom enhancement applied: brightness=" + brightness +
                       ", contrast=" + contrast + ", sharpness=" + sharpness +
                       ", noiseReduction=" + noiseReduction);
            return result;

        } catch (Exception e) {
            Log.e(TAG, "Error applying custom enhancement", e);
            return inputBitmap;
        }
    }

    /**
     * Save enhanced image to storage
     *
     * @param enhancedBitmap Enhanced bitmap to save
     * @param outputDirectory Output directory
     * @param prefix Filename prefix (e.g., "ENHANCED_", "BW_")
     * @return File path of saved image, or null if failed
     */
    public static String saveEnhancedImage(Bitmap enhancedBitmap,
                                          File outputDirectory,
                                          String prefix) {
        if (enhancedBitmap == null) {
            Log.e(TAG, "Enhanced bitmap is null");
            return null;
        }

        try {
            // Ensure output directory exists
            if (!outputDirectory.exists()) {
                outputDirectory.mkdirs();
            }

            // Generate filename with timestamp
            String timestamp = new SimpleDateFormat(FILENAME_FORMAT, Locale.US)
                    .format(new Date());
            String filename = prefix + timestamp + ".jpg";
            File outputFile = new File(outputDirectory, filename);

            // Save bitmap
            FileOutputStream out = new FileOutputStream(outputFile);
            enhancedBitmap.compress(Bitmap.CompressFormat.JPEG, 90, out);
            out.flush();
            out.close();

            String filePath = outputFile.getAbsolutePath();
            Log.d(TAG, "Enhanced image saved: " + filePath);
            return filePath;

        } catch (Exception e) {
            Log.e(TAG, "Error saving enhanced image", e);
            return null;
        }
    }

    /**
     * Quick enhance for document images
     * Applies a combination of filters optimized for documents
     *
     * @param inputBitmap Input bitmap
     * @return Enhanced bitmap
     */
    public static Bitmap quickDocumentEnhance(Bitmap inputBitmap) {
        if (inputBitmap == null) {
            Log.e(TAG, "Input bitmap is null");
            return null;
        }

        if (!OpenCVHelper.isInitialized()) {
            Log.e(TAG, "OpenCV not initialized");
            return inputBitmap;
        }

        try {
            Mat mat = OpenCVHelper.bitmapToMat(inputBitmap);

            // 1. Slight noise reduction
            Mat denoised = reduceNoise(mat, 3);
            mat.release();

            // 2. Auto-enhance contrast
            Mat enhanced = autoEnhance(denoised);
            denoised.release();

            // 3. Slight sharpening
            Mat sharpened = sharpenImage(enhanced, 3);
            enhanced.release();

            Bitmap result = OpenCVHelper.matToBitmap(sharpened);
            sharpened.release();

            Log.d(TAG, "Quick document enhancement applied");
            return result;

        } catch (Exception e) {
            Log.e(TAG, "Error in quick document enhance", e);
            return inputBitmap;
        }
    }

    /**
     * Get filter name for display
     *
     * @param filterType Filter type
     * @return Human-readable filter name
     */
    public static String getFilterName(FilterType filterType) {
        switch (filterType) {
            case AUTO_ENHANCE:
                return "Auto Enhance";
            case BLACK_AND_WHITE:
                return "Black & White";
            case BRIGHTNESS_CONTRAST:
                return "Brightness/Contrast";
            case NOISE_REDUCTION:
                return "Noise Reduction";
            case SHARPEN:
                return "Sharpen";
            case GRAYSCALE:
                return "Grayscale";
            case ORIGINAL:
            default:
                return "Original";
        }
    }
}
