package com.example.myapplication;

import android.content.Context;
import android.graphics.Bitmap;
import android.util.Log;
import android.widget.Toast;

import org.opencv.android.OpenCVLoader;
import org.opencv.android.Utils;
import org.opencv.core.CvType;
import org.opencv.core.Mat;
import org.opencv.core.MatOfPoint;
import org.opencv.core.MatOfPoint2f;
import org.opencv.core.Point;
import org.opencv.core.Rect;
import org.opencv.core.Scalar;
import org.opencv.core.Size;
import org.opencv.imgproc.Imgproc;

import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

/**
 * OpenCVHelper - Utility class for OpenCV initialization and image processing
 * Provides document scanning features like edge detection, perspective correction,
 * and image enhancement
 */
public class OpenCVHelper {

    private static final String TAG = "OpenCVHelper";
    private static boolean isOpenCVInitialized = false;

    /**
     * Initialize OpenCV library
     * Should be called once during app startup
     *
     * @param context Application or Activity context
     * @return true if initialization successful, false otherwise
     */
    public static boolean initOpenCV(Context context) {
        if (isOpenCVInitialized) {
            Log.d(TAG, "OpenCV already initialized");
            return true;
        }

        try {
            if (OpenCVLoader.initLocal()) {
                Log.d(TAG, "OpenCV initialization successful");
                isOpenCVInitialized = true;
                return true;
            } else {
                Log.e(TAG, "OpenCV initialization failed");
                Toast.makeText(context,
                    "OpenCV initialization failed",
                    Toast.LENGTH_SHORT).show();
                return false;
            }
        } catch (Exception e) {
            Log.e(TAG, "Error initializing OpenCV", e);
            Toast.makeText(context,
                "Error initializing image processing: " + e.getMessage(),
                Toast.LENGTH_SHORT).show();
            return false;
        }
    }

    /**
     * Check if OpenCV is initialized
     *
     * @return true if OpenCV is ready to use
     */
    public static boolean isInitialized() {
        return isOpenCVInitialized;
    }

    /**
     * Convert Android Bitmap to OpenCV Mat
     *
     * @param bitmap Input bitmap
     * @return OpenCV Mat object
     */
    public static Mat bitmapToMat(Bitmap bitmap) {
        Mat mat = new Mat(bitmap.getHeight(), bitmap.getWidth(), CvType.CV_8UC4);
        Utils.bitmapToMat(bitmap, mat);
        return mat;
    }

    /**
     * Convert OpenCV Mat to Android Bitmap
     *
     * @param mat Input Mat
     * @return Android Bitmap
     */
    public static Bitmap matToBitmap(Mat mat) {
        Bitmap bitmap = Bitmap.createBitmap(mat.cols(), mat.rows(), Bitmap.Config.ARGB_8888);
        Utils.matToBitmap(mat, bitmap);
        return bitmap;
    }

    /**
     * Detect document edges in an image
     *
     * @param inputMat Input image as Mat
     * @return List of corner points (4 points) or null if not found
     */
    public static List<Point> detectDocumentEdges(Mat inputMat) {
        try {
            Mat gray = new Mat();
            Mat blurred = new Mat();
            Mat edged = new Mat();

            // Convert to grayscale
            Imgproc.cvtColor(inputMat, gray, Imgproc.COLOR_BGR2GRAY);

            // Apply Gaussian blur to reduce noise
            Imgproc.GaussianBlur(gray, blurred, new Size(5, 5), 0);

            // Edge detection using Canny
            Imgproc.Canny(blurred, edged, 75, 200);

            // Find contours
            List<MatOfPoint> contours = new ArrayList<>();
            Mat hierarchy = new Mat();
            Imgproc.findContours(edged, contours, hierarchy,
                Imgproc.RETR_LIST, Imgproc.CHAIN_APPROX_SIMPLE);

            // Sort contours by area (largest first)
            Collections.sort(contours, new Comparator<MatOfPoint>() {
                @Override
                public int compare(MatOfPoint c1, MatOfPoint c2) {
                    double area1 = Imgproc.contourArea(c1);
                    double area2 = Imgproc.contourArea(c2);
                    return Double.compare(area2, area1);
                }
            });

            // Find the largest quadrilateral contour
            for (MatOfPoint contour : contours) {
                double peri = Imgproc.arcLength(new MatOfPoint2f(contour.toArray()), true);
                MatOfPoint2f approx = new MatOfPoint2f();
                Imgproc.approxPolyDP(new MatOfPoint2f(contour.toArray()), approx, 0.02 * peri, true);

                // Check if it's a quadrilateral
                if (approx.total() == 4) {
                    List<Point> points = new ArrayList<>();
                    for (int i = 0; i < 4; i++) {
                        points.add(new Point(approx.get(i, 0)));
                    }

                    // Release temporary mats
                    gray.release();
                    blurred.release();
                    edged.release();
                    hierarchy.release();

                    return orderPoints(points);
                }
            }

            // Release mats
            gray.release();
            blurred.release();
            edged.release();
            hierarchy.release();

            return null;
        } catch (Exception e) {
            Log.e(TAG, "Error detecting document edges", e);
            return null;
        }
    }

    /**
     * Order points in clockwise order: top-left, top-right, bottom-right, bottom-left
     *
     * @param points List of 4 corner points
     * @return Ordered list of points
     */
    private static List<Point> orderPoints(List<Point> points) {
        // Sort by y-coordinate (top two have smaller y)
        Collections.sort(points, new Comparator<Point>() {
            @Override
            public int compare(Point p1, Point p2) {
                return Double.compare(p1.y, p2.y);
            }
        });

        List<Point> orderedPoints = new ArrayList<>();

        // Top two points
        Point tl = points.get(0).x < points.get(1).x ? points.get(0) : points.get(1);
        Point tr = points.get(0).x > points.get(1).x ? points.get(0) : points.get(1);

        // Bottom two points
        Point bl = points.get(2).x < points.get(3).x ? points.get(2) : points.get(3);
        Point br = points.get(2).x > points.get(3).x ? points.get(2) : points.get(3);

        orderedPoints.add(tl);  // Top-left
        orderedPoints.add(tr);  // Top-right
        orderedPoints.add(br);  // Bottom-right
        orderedPoints.add(bl);  // Bottom-left

        return orderedPoints;
    }

    /**
     * Apply perspective transform to extract document
     *
     * @param inputMat Input image
     * @param corners 4 corner points of document
     * @return Transformed (warped) image
     */
    public static Mat applyPerspectiveTransform(Mat inputMat, List<Point> corners) {
        try {
            if (corners == null || corners.size() != 4) {
                return inputMat;
            }

            // Calculate dimensions for output
            double width = Math.max(
                distance(corners.get(0), corners.get(1)),
                distance(corners.get(2), corners.get(3))
            );

            double height = Math.max(
                distance(corners.get(0), corners.get(3)),
                distance(corners.get(1), corners.get(2))
            );

            // Destination points (rectangle)
            MatOfPoint2f dst = new MatOfPoint2f(
                new Point(0, 0),
                new Point(width - 1, 0),
                new Point(width - 1, height - 1),
                new Point(0, height - 1)
            );

            // Source points (document corners)
            MatOfPoint2f src = new MatOfPoint2f(
                corners.get(0), corners.get(1), corners.get(2), corners.get(3)
            );

            // Get perspective transform matrix
            Mat transformMatrix = Imgproc.getPerspectiveTransform(src, dst);

            // Apply transformation
            Mat warped = new Mat();
            Imgproc.warpPerspective(inputMat, warped, transformMatrix,
                new Size(width, height));

            return warped;
        } catch (Exception e) {
            Log.e(TAG, "Error applying perspective transform", e);
            return inputMat;
        }
    }

    /**
     * Calculate Euclidean distance between two points
     */
    private static double distance(Point p1, Point p2) {
        double dx = p1.x - p2.x;
        double dy = p1.y - p2.y;
        return Math.sqrt(dx * dx + dy * dy);
    }

    /**
     * Enhance document image (increase contrast, brightness)
     *
     * @param inputMat Input image
     * @return Enhanced image
     */
    public static Mat enhanceDocument(Mat inputMat) {
        try {
            Mat enhanced = new Mat();

            // Convert to grayscale if needed
            if (inputMat.channels() > 1) {
                Imgproc.cvtColor(inputMat, enhanced, Imgproc.COLOR_BGR2GRAY);
            } else {
                enhanced = inputMat.clone();
            }

            // Apply adaptive threshold for better text visibility
            Imgproc.adaptiveThreshold(enhanced, enhanced, 255,
                Imgproc.ADAPTIVE_THRESH_GAUSSIAN_C,
                Imgproc.THRESH_BINARY, 11, 2);

            return enhanced;
        } catch (Exception e) {
            Log.e(TAG, "Error enhancing document", e);
            return inputMat;
        }
    }

    /**
     * Apply bilateral filter to reduce noise while preserving edges
     *
     * @param inputMat Input image
     * @return Filtered image
     */
    public static Mat applyBilateralFilter(Mat inputMat) {
        try {
            Mat filtered = new Mat();
            Imgproc.bilateralFilter(inputMat, filtered, 9, 75, 75);
            return filtered;
        } catch (Exception e) {
            Log.e(TAG, "Error applying bilateral filter", e);
            return inputMat;
        }
    }

    /**
     * Draw document edges on image for visualization
     *
     * @param inputMat Input image
     * @param corners 4 corner points
     * @return Image with edges drawn
     */
    public static Mat drawDocumentEdges(Mat inputMat, List<Point> corners) {
        try {
            if (corners == null || corners.size() != 4) {
                return inputMat;
            }

            Mat output = inputMat.clone();

            // Draw lines connecting corners
            Scalar color = new Scalar(0, 255, 0); // Green
            int thickness = 3;

            for (int i = 0; i < 4; i++) {
                Point p1 = corners.get(i);
                Point p2 = corners.get((i + 1) % 4);
                Imgproc.line(output, p1, p2, color, thickness);
            }

            // Draw circles at corners
            for (Point corner : corners) {
                Imgproc.circle(output, corner, 10, new Scalar(255, 0, 0), -1);
            }

            return output;
        } catch (Exception e) {
            Log.e(TAG, "Error drawing document edges", e);
            return inputMat;
        }
    }

    /**
     * Resize image while maintaining aspect ratio
     *
     * @param inputMat Input image
     * @param maxDimension Maximum width or height
     * @return Resized image
     */
    public static Mat resizeImage(Mat inputMat, int maxDimension) {
        try {
            int width = inputMat.width();
            int height = inputMat.height();

            if (width <= maxDimension && height <= maxDimension) {
                return inputMat;
            }

            double scale = Math.min(
                (double) maxDimension / width,
                (double) maxDimension / height
            );

            int newWidth = (int) (width * scale);
            int newHeight = (int) (height * scale);

            Mat resized = new Mat();
            Imgproc.resize(inputMat, resized, new Size(newWidth, newHeight));

            return resized;
        } catch (Exception e) {
            Log.e(TAG, "Error resizing image", e);
            return inputMat;
        }
    }

    /**
     * Convert image to grayscale
     *
     * @param inputMat Input image
     * @return Grayscale image
     */
    public static Mat convertToGrayscale(Mat inputMat) {
        try {
            if (inputMat.channels() == 1) {
                return inputMat;
            }

            Mat gray = new Mat();
            Imgproc.cvtColor(inputMat, gray, Imgproc.COLOR_BGR2GRAY);
            return gray;
        } catch (Exception e) {
            Log.e(TAG, "Error converting to grayscale", e);
            return inputMat;
        }
    }
}
